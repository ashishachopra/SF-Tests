-- ============================================
-- EXECUTION REPORT FOR SP_CLUSTER_DETECTION_OPTIMIZED
-- ============================================

-- Summary Statistics (Last 7 Days)
SELECT 
    COUNT(*) AS "Number of Executions",
    ROUND(AVG(TOTAL_ELAPSED_TIME) / 1000, 2) AS "Avg Time (s)",
    ROUND(MIN(TOTAL_ELAPSED_TIME) / 1000, 2) AS "Min Time (s)",
    ROUND(MAX(TOTAL_ELAPSED_TIME) / 1000, 2) AS "Max Time (s)",
    ROUND(SUM(CREDITS_USED_CLOUD_SERVICES), 4) AS "Credits Consumed"
FROM TABLE(A01A0E_GBU_FINCRIME_POC.INFORMATION_SCHEMA.QUERY_HISTORY())
WHERE QUERY_TEXT ILIKE '%SP_CLUSTER_DETECTION_OPTIMIZED%'
  AND QUERY_TEXT ILIKE '%CALL%'
  AND WAREHOUSE_NAME = 'GBU_A01A0E_FINCRIME_DE_POC_WH_4XL'
  AND EXECUTION_STATUS = 'SUCCESS';

-- Detailed Execution History
SELECT 
    QUERY_ID,
    START_TIME,
    END_TIME,
    ROUND(TOTAL_ELAPSED_TIME / 1000, 2) AS "Elapsed Time (s)",
    EXECUTION_STATUS,
    ROUND(CREDITS_USED_CLOUD_SERVICES, 6) AS "Credits Used",
    USER_NAME,
    WAREHOUSE_NAME
FROM TABLE(A01A0E_GBU_FINCRIME_POC.INFORMATION_SCHEMA.QUERY_HISTORY())
WHERE QUERY_TEXT ILIKE '%SP_CLUSTER_DETECTION_OPTIMIZED%'
  AND QUERY_TEXT ILIKE '%CALL%'
  AND WAREHOUSE_NAME = 'GBU_A01A0E_FINCRIME_DE_POC_WH_4XL'
  AND EXECUTION_STATUS = 'SUCCESS'
ORDER BY START_TIME DESC;
